\section*{Question 1}

Evaluate the limit:
\[
\lim_{x \to a} \frac{x^x - x^a}{a^x - a^a}
\]

Let us define:
\[
f(x) = x^x = e^{x \log x}
\]
Then compute derivatives:
\[
\frac{d}{dx}(x^x) = x^x(\log x + 1)
\]

Using L'Hôpital's Rule (since both numerator and denominator tend to 0):
\[
\lim_{x \to a} \frac{x^x - x^a}{a^x - a^a} = \frac{d}{dx}(x^x - x^a)\Big|_{x = a} \Bigg/ \frac{d}{dx}(a^x - a^a)\Big|_{x = a}
\]

\[
= \frac{x^x(\log x + 1)\big|_{x = a}}{a^x \log a\big|_{x = a}} = \frac{a^a(\log a + 1)}{a^a \log a} = \frac{\log a + 1}{\log a}
\]

\section*{Question 2}

An orthonormal basis in $\mathbb{R}^n$ is a set of $n$ vectors that are:
\begin{itemize}
    \item Unit vectors: $\|v_i\| = 1$
    \item Mutually orthogonal: $v_i \cdot v_j = 0$ for $i \neq j$
\end{itemize}

\subsection*{How many vectors?}

There are $n$ vectors in the basis because the dimension of $\mathbb{R}^n$ is $n$.

\subsection*{Sets A and B}

If $A$ and $B$ are disjoint subsets of the orthonormal basis:
\begin{itemize}
    \item The vectors in $A$ cannot be linearly dependent on any vector in $B$ because orthonormal vectors are all linearly independent.
    \item Therefore, no vector in $A$ lies in the span of $B$.
\end{itemize}

\subsection*{Matrix $P$ and $P^\top P$}

Let $A$ be an $n \times k$ matrix of orthonormal columns (with $k \leq n$) and $0$ be an $n \times (n - k)$ zero matrix.

\[
P = [A\ 0]
\Rightarrow P^\top P =
\begin{bmatrix}
A^\top A & 0 \\
0 & 0
\end{bmatrix}
\]

Since $A$ has orthonormal columns:
\[
A^\top A = I_k \Rightarrow P^\top P = 
\begin{bmatrix}
I_k & 0 \\
0 & 0
\end{bmatrix}
\]

\section*{Question 3}

Given:
\[
E[u] = \sum_{i=1}^{1} \sum_{j=0}^{1} \cos(u[i, j]) \sin(u[i-1, j])
\]

Since $u$ is a $2 \times 2$ matrix with indices from $0$ to $1$, this becomes:
\[
E[u] = \cos(u[1,0]) \sin(u[0,0]) + \cos(u[1,1]) \sin(u[0,1])
\]

To compute $\nabla_u E$, apply the chain rule:

\[
\frac{\partial E}{\partial u[1,0]} = -\sin(u[1,0]) \sin(u[0,0])
\]
\[
\frac{\partial E}{\partial u[0,0]} = \cos(u[1,0]) \cos(u[0,0])
\]
\[
\frac{\partial E}{\partial u[1,1]} = -\sin(u[1,1]) \sin(u[0,1])
\]
\[
\frac{\partial E}{\partial u[0,1]} = \cos(u[1,1]) \cos(u[0,1])
\]

\[
\Rightarrow \nabla_u E =
\begin{bmatrix}
\cos(u[1,0]) \cos(u[0,0]) & \cos(u[1,1]) \cos(u[0,1]) \\
-\sin(u[1,0]) \sin(u[0,0]) & -\sin(u[1,1]) \sin(u[0,1])
\end{bmatrix}
\]

\section*{Question 4}

\subsection*{Poisson Distribution PDF}

The probability mass function of the Poisson distribution:
\[
p(x; \lambda) = \frac{e^{-\lambda} \lambda^x}{x!}, \quad x \in \mathbb{N}_0
\]

\subsection*{Expected Value}

\[
\mathbb{E}[x] = \sum_{x=0}^{\infty} x \cdot \frac{e^{-\lambda} \lambda^x}{x!}
= e^{-\lambda} \sum_{x=1}^{\infty} \frac{x \lambda^x}{x!}
= e^{-\lambda} \sum_{x=1}^{\infty} \frac{\lambda^x}{(x-1)!}
= \lambda
\]

\subsection*{Maximum Likelihood Estimate (MLE)}

Given samples $x_1, x_2, ..., x_m$, the log-likelihood is:
\[
\log L(\lambda) = \sum_{i=1}^m \log p(x_i; \lambda)
= \sum_{i=1}^m \left(-\lambda + x_i \log \lambda - \log x_i!\right)
\]

\[
\frac{d}{d\lambda} \log L(\lambda) = -m + \frac{1}{\lambda} \sum x_i = 0
\Rightarrow \hat{\lambda}_{MLE} = \frac{1}{m} \sum_{i=1}^m x_i
\]

Hence, the MLE of $\lambda$ is the **sample mean**.
